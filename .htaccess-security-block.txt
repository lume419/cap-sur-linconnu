# ============================================================================
# BLOC À AJOUTER à votre .htaccess existant sur le serveur (o2switch) — pas à
# copier tel quel en écrasant le fichier ! cPanel a très probablement généré
# un .htaccess contenant des directives Passenger (PassengerAppRoot,
# PassengerStartupFile, PassengerNodejs...) qui font tourner l'app Node — les
# retirer casserait le site. Ouvrez le .htaccess existant du site (Gestionnaire
# de fichiers cPanel, ou FTP/SFTP) et collez ce bloc à la fin, sans rien
# supprimer.
#
# Faites une copie de sauvegarde du fichier existant avant de le modifier.
# ============================================================================

# --- Redirection HTTP -> HTTPS ---
RewriteEngine On
RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]

# --- Bloque l'accès direct aux fichiers techniques du serveur ---
# Confirmé exposés publiquement au moment de l'audit (18/08/2026) :
# https://road.lume419.fr/server.js et /package.json répondaient tous les
# deux 200 avec le code source brut. Ces fichiers ne sont jamais nécessaires
# au navigateur — Node/Passenger les lit directement sur le disque du
# serveur, sans passer par une requête HTTP, donc cette règle ne bloque en
# rien le fonctionnement de l'app elle-même, seulement leur consultation
# publique. Redirige vers l'accueil plutôt qu'un 403/404, pour ne pas
# confirmer leur existence à quelqu'un qui devinerait l'URL.
RewriteRule ^(server\.js|package(-lock)?\.json|README\.md)$ /? [R=301,L]

# --- Dossier .git éventuel et node_modules ---
# .git exposerait tout l'historique du dépôt (déjà bloqué en 403 au moment de
# l'audit, mais gardé ici en filet de sécurité) ; node_modules n'est jamais
# utile au navigateur.
RewriteRule ^\.git(/.*)?$ /? [R=301,L]
RewriteRule ^node_modules(/.*)?$ /? [R=301,L]

# --- Dossiers du dépôt servis tels quels par Apache (audit du 17/09/2026) ---
# Apache sert la RACINE du dépôt : tout fichier réel y est renvoyé sans passer
# par Node. Constaté en production : /lib/trip-engine.js, /scripts/…,
# /cache/search-index/… (index de recherche de plusieurs centaines de Mo,
# téléchargeable par plages), /data/charging-stations.txt, /data/hiking.json,
# /tmp/restart.txt et /public/… répondaient 200. Le site n'utilise AUCUNE de
# ces URL : les pages et scripts du navigateur sont servis par Node depuis
# public/ à la racine du site (/js/app.js, /css/…), jamais sous /public/.
RewriteRule ^(lib|scripts|cache|tmp|public|data|\.claude)(/.*)?$ /? [R=301,L]
RewriteRule ^(\.htaccess-security-block\.txt|\.gitignore|\.gitattributes|\.env.*)$ /? [R=301,L]
